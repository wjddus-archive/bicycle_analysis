------------------------------------------------------------------------------
1. 코드 생성을 위한 프롬프트 
------------------------------------------------------------------------------

너는 Python, SQLite, Streamlit에 능통한 시니어 개발자이면서, 코딩을 처음 배우는 사람에게도 쉽게 
설명할 수 있는 특별한 재능을 가지고 있어. 공공데이터 분석 대시보드를 Streamlit으로 만들기 위한 
코드(app.py, requirements.txt)를 작성해줘. 초보자도 그대로 실행할 수 있도록 과정을 자세히 설명해줘.

[데이터]
bicycle.db (SQLite)가 같은 폴더에 있고, 스키마는 다음과 같아.
대여소(대여소번호 PK, 보관소명, 자치구, 상세주소, 위도, 경도, 설치시기, LCD, QR, 운영방식)
이용정보(id PK, 대여일자[YYYYMM], 대여소번호, 대여소명, 대여구분코드, 성별, 연령대코드, 이용건수, 
운동량, 탄소량, 이동거리, 이용시간)
기온(년월[YYYYMM], 지점, 평균기온, 평균최저기온, 평균최고기온, PK=년월+지점) 
강수량(년월[YYYYMM], 지점, 강수량, PK=년월+지점)

[차트 3개 만들기]
각 차트마다 ① 시각화, ② 사용한 SQL, ③ 인사이트(2~3줄) 를 함께 보여줘. 
1. 이용자 패턴 분석 (Demographics)
연령대별 이용 비중 — 이용정보의 연령대코드별 총 이용건수 (도넛 차트)
권종별 이용 패턴 — 이용정보의 대여구분코드별 총 이용건수 (파이 차트)
성별 이용 현황 — 이용정보의 성별별 총 이용건수 (막대 차트)
2. 지역 및 인프라 분석 (Location)
자치구별 이용량 — 이용정보와 대여소를 JOIN, 자치구별 총 이용건수 (가로 막대 차트)
운영방식별 선호도 — 이용정보와 대여소를 JOIN, 운영방식(QR/LCD)별 총 이용건수 (막대 차트)
대여소 규모별 이용량 — 대여소의 거치대 수(QR+LCD) 대비 총 이용건수 (산점도)
3. 이용 효율 및 환경 분석 (Efficiency)
이동거리 분포 — 이용정보의 이동거리 구간별 이용건수 (히스토그램)
월별 탄소 절감량 — 이용정보의 대여일자별 탄소량 합계 (라인 차트)
운동 효율 분석 — 이용정보의 연령대별 평균 운동량 (막대 차트)

3. 테이블 정보 (Create Table 구문)
------------------------------------------------------------------------------
CREATE TABLE 대여소 (
    대여소번호      TEXT        NOT NULL,   -- '00102' 등 0으로 시작
    보관소명        TEXT,
    자치구          TEXT,
    상세주소        TEXT,
    위도            REAL,
    경도            REAL,
    설치시기        TEXT,                   -- 'YYYY-MM-DD'
    LCD             INTEGER,
    QR              INTEGER,
    운영방식        TEXT,
    PRIMARY KEY (대여소번호)
);
CREATE TABLE 이용정보 (
    id              INTEGER     PRIMARY KEY,
    대여일자        TEXT        NOT NULL,   -- 년월 '202507'
    대여소번호      TEXT        NOT NULL,
    대여소명        TEXT,
    대여구분코드    TEXT,                   -- 정기권, 일일권 등
    성별            TEXT,                   -- M, F, 빈값
    연령대코드      TEXT,                   -- 20대, 30대 등
    이용건수        INTEGER,
    운동량          REAL,
    탄소량          REAL,
    이동거리        REAL,
    이용시간        INTEGER)
);
CREATE TABLE 기온 (
    년월            TEXT        NOT NULL,   -- '202507'
    지점            TEXT        NOT NULL,   -- '108'
    평균기온        REAL,
    평균최저기온    REAL,
    평균최고기온    REAL,
    PRIMARY KEY (년월, 지점)
);
CREATE TABLE 강수량 (
    년월            TEXT        NOT NULL,   -- '202507'
    지점            TEXT        NOT NULL,   -- '108'
    강수량          REAL,
    PRIMARY KEY (년월, 지점)
);
